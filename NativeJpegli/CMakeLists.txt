cmake_minimum_required(VERSION 3.20)
project(PhotoJpegli LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option TOOLS DEVTOOLS JNI MANPAGES DOXYGEN BENCHMARK SJPEG OPENEXR TCMALLOC JPEGLI_LIBJPEG)
  set(JPEGLI_ENABLE_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(JPEGLI_ENABLE_SKCMS ON CACHE BOOL "" FORCE)
set(JPEGLI_BUNDLE_LIBPNG OFF CACHE BOOL "" FORCE)
set(JPEGLI_STATIC ON CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_subdirectory("${JPEGLI_SOURCE}" upstream EXCLUDE_FROM_ALL)
add_library(PhotoJpegli STATIC bridge.cpp)
target_include_directories(PhotoJpegli PUBLIC ../Sources/JpegliBridge/include)
target_include_directories(PhotoJpegli PRIVATE "${JPEGLI_SOURCE}")
target_link_libraries(PhotoJpegli PUBLIC jpegli-static)
add_executable(photo-jpegli-check compatibility.cpp)
target_link_libraries(photo-jpegli-check PRIVATE PhotoJpegli)
